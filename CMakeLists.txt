target_compile_features(glint_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(glint_core PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_compile_definitions(glint_core PRIVATE _CRT_SECURE_NO_WARNINGS=1)
  set_property(TARGET glint_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(glint_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Keep a hung native test from consuming the entire CI job. Individual
# integration tests below retain their shorter timeout.
if(GLINT_BUILD_TESTS)
  get_property(GLINT_REGISTERED_TESTS DIRECTORY PROPERTY TESTS)
  if(GLINT_REGISTERED_TESTS)
    set_tests_properties(${GLINT_REGISTERED_TESTS} PROPERTIES TIMEOUT 120)
    if(WIN32)
      set_tests_properties(glint_pipe_server PROPERTIES TIMEOUT 20)
      set_tests_properties(glint_native_host PROPERTIES TIMEOUT 45)
      set_tests_properties(glint_update_process PROPERTIES TIMEOUT 180)
      # Desktop geometry, keyboard focus and global hotkeys are validated on
      # the developer desktop; CI excludes this label in its headless checks.
      set_tests_properties(glint_search_navigation glint_search_caption glint_hotkey glint_hotkey_window
        glint_setup_window PROPERTIES LABELS desktop)
    endif()
  endif()
endif()
